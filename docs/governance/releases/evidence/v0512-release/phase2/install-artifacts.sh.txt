#!/bin/bash
set -euo pipefail
set +x
[ "$(id -un)" = deploy ]
[ "$(hostname)" = haiyue-vps ]
staging=/opt/cps-novel/shared/artifacts/staging
final=84af0d7ef5ef36b06fa9783c114fe831adab4e13
previous=eb40756416b186ca4b293ff57e24026af78cf4f9
release=/opt/cps-novel/releases/$final
cd "$staging"
printf '%s  %s\n' ed35a77cbefe12930cf166400d9d228bbf2ff98675fe2b909a28f3ebbde0e7ca cps-novel-0.5.12-84af0d7.tar.zst | sha256sum -c -
sha256sum cps-novel-0.5.12-84af0d7.tar.zst
printf '%s  %s\n' 5162e2485dea411835a5e241453e0095ca74d888fbd2f5e8f8e6d945687169c3 cps-novel-v0.5.12.bundle | sha256sum -c -
printf '%s  %s\n' a441e1242dbb66729dbc1a9a463ae41c608d576d8e5089c05c15077d18a6488a "$final.json" | sha256sum -c -
if test -e "$release"; then echo RELEASE_DIRECTORY_EXISTS; exit 65; fi
git clone --no-hardlinks "/opt/cps-novel/releases/$previous" "$release" </dev/null
git -C "$release" fetch "$staging/cps-novel-v0.5.12.bundle" refs/heads/release/v0.5.12-2026-10-09 </dev/null
git -C "$release" checkout --detach "$final" </dev/null
[ "$(git -C "$release" rev-parse HEAD)" = "$final" ]
[ "$(git -C "$release" rev-parse HEAD^{tree})" = 0367259156c20d30d2db02026c87827a4914bc97 ]
[ -z "$(git -C "$release" status --porcelain=v1)" ]
echo CODE_FINAL_TREE_CLEAN=PASS
"$release/scripts/preproduction/verify-release-archive.sh" --manifest "$staging/$final.json" --approved-commit "$final" --expected-archive-sha256 ed35a77cbefe12930cf166400d9d228bbf2ff98675fe2b909a28f3ebbde0e7ca --load </dev/null
