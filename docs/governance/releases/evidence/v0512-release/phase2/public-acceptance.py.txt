class Parse(HTMLParser):
 def __init__(self):super().__init__(convert_charrefs=True);self.stack=[];self.h1=[];self.title=[];self.meta={};self.text=[];self.pagination=[]
 def handle_starttag(self,tag,attrs):
  a=dict(attrs)
  if tag=='meta':self.meta[a.get('name',a.get('property',''))]=a.get('content','')
  if tag not in ['meta','link','img','input','br','hr','source','wbr']:self.stack.append((tag,a))
 def handle_endtag(self,tag):
  for i in range(len(self.stack)-1,-1,-1):
   if self.stack[i][0]==tag:self.stack=self.stack[:i];break
 def handle_data(self,data):
  if any(t in ['script','style'] for t,a in self.stack):return
  self.text.append(data)
  if any(t=='h1' for t,a in self.stack):self.h1.append(data)
  if any(t=='title' for t,a in self.stack):self.title.append(data)
  if any(a.get('data-testid')=='pagination' for t,a in self.stack):self.pagination.append(data)
