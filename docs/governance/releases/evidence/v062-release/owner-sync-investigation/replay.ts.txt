import fs from 'node:fs';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
import { parseNovelCountryReport, parseNovelReportRows } from './src/lib/adapters/moboreader-revenue-parser.ts';

const base='.tmp/v062-release/country-investigation';
const input=JSON.parse(fs.readFileSync(base+'/snapshots.json','utf8'));
const countrySnapshots=input.snapshots.filter(s=>s.dimension.startsWith('country:'));
const parsed=countrySnapshots.map(s=>({...parseNovelCountryReport(s.key,s.payload),dimension:s.dimension,snapshotId:s.id,payloadSha256:s.payloadSha256}));
assert.equal(parsed.length,15);
assert.ok(parsed.every(p=>p.droppedRowCount===0&&p.duplicateRowCount===0&&p.countrySum===p.total));
const rows=new Map();
for (const p of parsed) for (const country of p.countries) {
  const key=JSON.stringify([p.date,country.label]);
  const row=rows.get(key)??{date:p.date,label:country.label,code:country.code,newUsers:0,activeUsers:0};
  if(p.dimension==='country:0')row.activeUsers+=country.users;
  else row.newUsers+=country.users;
  rows.set(key,row);
}
assert.equal(rows.size,input.countries.length);
for(const stored of input.countries){
  assert.deepEqual(rows.get(JSON.stringify([stored.date,stored.label])),{date:stored.date,label:stored.label,code:stored.code,newUsers:stored.newUsers,activeUsers:stored.activeUsers});
}
const daily=parseNovelReportRows(input.snapshots.filter(s=>s.dimension==='1').map(s=>s.payload));
assert.equal(daily.detail.length,input.daily.length);
assert.equal(daily.duplicateDateCount,0);
assert.equal(daily.droppedRowCount,0);
for(const row of daily.detail){
  const stored=input.daily.find(s=>s.date===row.date);
  assert.ok(stored);
  assert.equal(row.activeUsers,stored.activeUsers);
  assert.equal(row.newUsers,stored.newUsers);
}
const comparisons=daily.detail.map(d=>{
  const active=parsed.find(p=>p.date===d.date&&p.dimension==='country:0');
  const nu=parsed.find(p=>p.date===d.date&&p.dimension==='country:1');
  return {date:d.date,dailyRawNewUsers:d.newUsers,countryRawNewUsersTotal:nu?.total??0,countryRawNewUsersSum:nu?.countrySum??0,dailyRawActiveUsers:d.activeUsers,countryRawActiveUsersTotal:active?.total??0,countryRawActiveUsersSum:active?.countrySum??0};
}).sort((a,b)=>a.date.localeCompare(b.date));
const mismatches=comparisons.filter(d=>d.dailyRawNewUsers!==d.countryRawNewUsersTotal||d.dailyRawActiveUsers!==d.countryRawActiveUsersTotal);
assert.equal(mismatches.length,4);
const logs=JSON.parse(fs.readFileSync(base+'/logs.json','utf8'));
assert.equal(logs.observations.length,17);
assert.ok(logs.observations.every(o=>o.httpStatus===200&&o.outcome==='ok'));
