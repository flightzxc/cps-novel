 out={'at':x['at'],'wave':wave,'mode':mode,'generation':s['generation'],'generationTasks':s['generationTasks'],'drafts':s['drafts'],'published':s['published'],'indexnow':s['indexnow'],'preview':s['preview'],'mainPreviewBacklog':b,'sitemap':s['sitemap'],'readyForNextWave':s['readyForNextWave'],'stopReasons':reasons,'logs':x['logs']}
 print(json.dumps(out),flush=True)

  p=root/'STOP.json'
  if not p.exists():p.write_text(json.dumps(out,indent=2)+'\n');p.chmod(0o600)
  sys.exit(2)
 if mode=='generation' and s['generation']['notSubmitted']==0 and s['generation']['pending']==0 and all(t and t['status'] in ['completed','completed_with_errors','failed','cancelled'] for t in s['generationTasks']):sys.exit(0)
 if mode=='closing' and s['readyForNextWave']:sys.exit(0)
