BEGIN ISOLATION LEVEL REPEATABLE READ READ ONLY;
DO $$ BEGIN
IF EXISTS (SELECT 1 FROM generic_task b WHERE b.status IN ('pending','processing') AND EXISTS (SELECT 1 FROM generic_task s WHERE s.parent_task_id=b.id AND s.task_type='promo_link.claim.v1')) OR EXISTS (SELECT 1 FROM generic_task WHERE task_type='promo_link.claim.v1' AND status IN ('pending','processing')) OR EXISTS (SELECT 1 FROM side_effect_intent WHERE status IN ('prepared','claim_retry_blocked')) OR EXISTS (SELECT 1 FROM generic_task_item WHERE status='processing') OR EXISTS (SELECT 1 FROM channel_sync_task_item WHERE status='processing') THEN RAISE EXCEPTION 'PUBLIC_CUTOVER_PAUSE_GATE=FAIL'; END IF;
IF EXISTS (SELECT 1 FROM generic_task WHERE (task_type LIKE 'article.publish.%' OR task_type LIKE 'article.generate.%') AND status IN ('pending','processing')) THEN RAISE EXCEPTION 'ARTICLE_INFLIGHT=FAIL'; END IF;
IF EXISTS (SELECT 1 FROM generic_task WHERE task_type='moboreader.preview_refresh.v1' AND status IN ('pending','processing')) THEN RAISE EXCEPTION 'PREVIEW_INFLIGHT=FAIL'; END IF;
IF EXISTS (SELECT 1 FROM generic_task WHERE task_type='changdu.revenue_sync.v1' AND status IN ('pending','processing')) THEN RAISE EXCEPTION 'REVENUE_INFLIGHT=FAIL'; END IF;
END $$;
SELECT 'PUBLIC_CUTOVER_PAUSE_GATE=PASS';
SELECT jsonb_build_object(
'migrations',(SELECT jsonb_agg(jsonb_build_object('name',migration_name,'checksum',checksum,'finished',finished_at IS NOT NULL,'rolled_back',rolled_back_at IS NOT NULL) ORDER BY migration_name) FROM _prisma_migrations),
'inflight',(SELECT coalesce(jsonb_agg(x),'[]'::jsonb) FROM (SELECT task_type,status,count(*) FROM generic_task WHERE (task_type LIKE 'article.publish.%' OR task_type LIKE 'article.generate.%' OR task_type='changdu.revenue_sync.v1') AND status IN ('pending','processing') GROUP BY task_type,status) x),
'tableCount',(SELECT count(*) FROM pg_tables WHERE schemaname='public' AND tablename<>'_prisma_migrations'),
'sampleBookPath',(SELECT '/novel/'||a.slug||'-p'||a.public_page_short_id FROM article a WHERE a.status='published' AND a.deleted_at IS NULL AND a.seo_visibility='public' AND a.locale='en' AND a.article_type='novel_article' AND EXISTS(SELECT 1 FROM novel_chapter c WHERE c.novel_id=a.novel_id AND c.canonical_chapter_number=1 AND c.deleted_at IS NULL) ORDER BY a.id LIMIT 1));
COMMIT;
