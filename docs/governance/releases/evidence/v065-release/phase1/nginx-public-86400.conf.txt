    ~*ClaudeBot ClaudeBot;

    ~*Bytespider Bytespider;
    ~*CCBot CCBot;
}
# Next 16.1.6 sends next-router-prefetch: 1 or 2. A nonempty value
# selects the prefetch budget; it never bypasses the crawler budget.
map $http_next_router_prefetch $cps_page_key { "" $server_name; default ""; }
map $http_next_router_prefetch $cps_prefetch_key { "" ""; default $server_name; }
map $http_next_router_prefetch $cps_prefetch_ip { "" ""; default $binary_remote_addr; }
map $host $cps_public_robots { default ""; }
map $host $cps_public_hsts { default "max-age=86400"; }
limit_conn_zone $cps_page_key zone=cps_edge_page_conn:10m;
limit_req_zone $cps_page_key zone=cps_edge_page_rate:10m rate=12r/s;
limit_conn_zone $cps_prefetch_key zone=cps_edge_prefetch_conn:10m;
limit_req_zone $cps_prefetch_ip zone=cps_edge_prefetch_rate:10m rate=5r/s;
limit_conn_zone $server_name zone=cps_edge_go_conn:10m;
limit_req_zone $binary_remote_addr zone=cps_edge_go_rate:10m rate=10r/s;
limit_req_zone $cps_training_bot zone=cps_edge_bot_page:10m rate=10r/m;
limit_req_zone $binary_remote_addr zone=cps_edge_search_ip:10m rate=20r/m;
limit_req_zone $server_name zone=cps_edge_search_all:1m rate=3r/s;
limit_req_zone $cps_training_bot zone=cps_edge_bot_go:10m rate=1r/m;
limit_req_zone $binary_remote_addr zone=cps_edge_static:10m rate=50r/s;
limit_req_zone $binary_remote_addr zone=cps_edge_admin_general:10m rate=30r/s;
limit_req_zone $binary_remote_addr zone=cps_edge_admin_login:10m rate=2r/s;
limit_req_zone $binary_remote_addr zone=cps_edge_admin_api:10m rate=20r/s;
limit_req_status 429;
limit_conn_status 429;
log_format cps_edge_json escape=json
    '{"time":"$time_iso8601","host":"$host","remote_addr":"$remote_addr",'
    '"request":"$request","status":$status,"limit_req_status":"$limit_req_status",'
    '"limit_conn_status":"$limit_conn_status","request_time":$request_time,'
    '"upstream_response_time":"$upstream_response_time","user_agent":"$http_user_agent"}';
gzip_types text/css application/javascript text/javascript application/json application/xml text/xml image/svg+xml text/plain;
gzip_vary on;
